Constructors
************
1. Constructor is a method.
2. Constructor is having the same name as the class name.
3. Constructor does not have any return type.


Advantage
*********
Constructor invokes/executed automatically, once the object is created.


Usage
*****
Constructor is used to initialize the default values to the class variables.




1. Implicit Constructor (JVM)
2. Explicit Constructor
   1. Default or Un-Parameterized Constructor
   2. Parameterized Constructor




package day2;

public class Demo1 {

	byte a;
	short b;
	int c;
	long d;
	float e;
	double f;
	boolean g;
	char h;
	String i;
	
	public Demo1() {
		a = 10;
		b = 20;
	}
	
	public static void main(String[] args) {

		Demo1 obj = new Demo1();		
		System.out.println("a = " + obj.a + "\nb = " + obj.b + "\nc = " + obj.c);
		System.out.println("d = " + obj.d + "\nb = " + obj.e + "\nc = " + obj.f);
		System.out.println("g = " + obj.g + "\nh = " + obj.h + "\ni = " + obj.i);
		System.out.println();
		
		Demo1 obj2 = new Demo1();		
		System.out.println("a = " + obj2.a + "\nb = " + obj2.b + "\nc = " + obj2.c);
		System.out.println("d = " + obj2.d + "\nb = " + obj2.e + "\nc = " + obj2.f);
		System.out.println("g = " + obj2.g + "\nh = " + obj2.h + "\ni = " + obj2.i);
		System.out.println();
		
		Demo1 obj3 = new Demo1();		
		System.out.println("a = " + obj3.a + "\nb = " + obj3.b + "\nc = " + obj3.c);

		System.out.println("g = " + obj3.g + "\nh = " + obj3.h + "\ni = " + obj3.i);
		System.out.println();
	}
}

package day2;

class Customer {	
	private int custId;
	private String custName;
	private double balance;
	
	//Right Click -> Source -> Generate Constructor using Super Class
	public Customer() {
	}

	//Right Click -> Source -> Generate Constructor using Feilds
	public Customer(int custId, String custName, double balance) {
		this.custId = custId;
		this.custName = custName;
		this.balance = balance;
	}

	//Right Click -> Source -> Generate Getters & Setters
	public int getCustId() {
		return custId;
	}
	public void setCustId(int custId) {
		this.custId = custId;
	}

	public String getCustName() {
		return custName;
	}
	public void setCustName(String custName) {
		this.custName = custName;
	}

	public double getBalance() {
		return balance;
	}
	public void setBalance(double balance) {
		this.balance = balance;
	}
	
	public void showCustomer() {
		System.out.println("Customer Id: " + custId);
		System.out.println("Customer Name: " + custName);
		System.out.println("Account Balance: " + balance + "\n");
	}	
}

public class CustomerDemo {
	public static void main(String[] args) {
		
		Customer cust1 = new Customer(1001, "Harsha", 1212.12);
		cust1.showCustomer();
		
	}
}

package day2;

class Customer {	
	
	private static int counter = 1001;
	
	private int custId;
	private String custName;
	private double balance;
	
	public Customer() {
		this.custId = counter++;
	}

	public Customer(String custName, double balance) {
		
		this.custId = counter++;
		this.custName = custName;
		this.balance = balance;
	}

	public int getCustId() {
		return custId;
	}
	public void setCustId(int custId) {
		this.custId = custId;
	}

	public String getCustName() {
		return custName;
	}
	public void setCustName(String custName) {
		this.custName = custName;
	}

	public double getBalance() {
		return balance;
	}
	public void setBalance(double balance) {
		this.balance = balance;
	}

	@Override
	public String toString() {
		return "Customer [custId=" + custId + ", custName=" + custName + ", balance=" + balance + "]";

public class CustomerDemo {
	public static void main(String[] args) {
		
		Customer[] customers = new Customer[3];
		
		customers[0] = new Customer("Harsha", 1212.12);
		customers[1] = new Customer("Pasha ", 2323.23);

		for (Customer cust : customers) {
			System.out.println(cust);
