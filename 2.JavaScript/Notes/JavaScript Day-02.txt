<html lang="en">
<head>
    <title>JavaScript</title>
    <script>   
        function myfun() {
            let num = document.getElementById("num").value;
            let count = 0;
            let result = "";

            for (let i = 2; i < num; i++) {
                if (num % i == 0) {
                    count++;
                    break;
                }
            }

            if (count == 0)
                result = "Prime Number";
            else
                result = "Not a Prime Number";


            document.getElementById("result").innerHTML = result;
        }       
    </script> 
</head>
<body>    
    Enter a Number: <input type="number" id="num" /> <br/>
    <button onclick="myfun()">Click Me</button> <br/>
    <h1 id="result"></h1>
</body>
</html>

    <title>JavaScript</title>
    <script> 
        function checkPrime(num) {
            for (let i = 2; i < num; i++) {
                if (num % i == 0)
                    return false;
            }
            return true;
        }    

        function myfun() {
            let start = parseInt(document.getElementById("start").value);
            let end = parseInt(document.getElementById("end").value);
            let result = "";
            let i = 1;
            end -= 1;
            
            for (i = start; i < end; i++) {
                if (checkPrime(i) && checkPrime(i + 2)) {
                    result += "(" + i + ", " + (i + 2) + ") <br/>";
                }
            }
                      
            document.getElementById("result").innerHTML = result;
        }       
    </script> 
</head>
<body>    
    Enter a Start Value: <input type="text" id="start" /> <br/>
    Enter a End Value:   <input type="text" id="end"   /> <br/>

    <button onclick="myfun()">Click Me</button> <br/>
    <h1 id="result"></h1>
</body>
