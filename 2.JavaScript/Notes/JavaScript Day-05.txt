Scope of Const, Let and Var Variables
*************************************
{
    const x = 10;
    let y = 20;
    var z = 30;

    console.log("const x = " + x);
    console.log("let y = " + y);
    console.log("var z = " + z);
}

//console.log("const x = " + x);    //Not Accessible
//console.log("let y = " + y);      //Not Accessible
console.log("var z = " + z);        //Accessible





Scope of var variable
*********************
{
    console.log("Outer Block Starts \n");
    {
        console.log("First Inner Block Starts \n");
        {
            console.log("Second Inner Block Starts");
            let a = 10;
            var b = 20;
            console.log("let a = " + a + ", var b = " + b);
            console.log("Second Inner Block Ends \n");
        }                
        console.log("var b = " + b);
        console.log("First Inner Block Ends \n");
    }
    console.log("var b = " + b);
    console.log("Outer Block Ends \n");
}

Demo on map() function
**********************

//Normal Approach
function findSquare(arr) {
    let newArr = [];
    for (let i = 0; i < arr.length; i++) {
        newArr[i] = arr[i] * arr[i];
    }
    return newArr;
}

let arr = [1, 2, 3, 4, 5]
let result = findSquare(arr);
console.log(result);




//Approach-2
let arr = [1, 2, 3, 4, 5]
let result = arr.map(function(num) {
    return num * num;
});
console.log(result);




//Approach-3
let arr = [1, 2, 3, 4, 5]
let result = arr.map((num) => num * num);
console.log(result);







Demo on filter() function
*************************

//Normal Approach
function findEven(arr) {
    let newArr = [];
    for (let i = 0; i < arr.length; i++) {
        if (arr[i] % 2 == 0) {
            newArr.push(arr[i]);
        }
    }
    return newArr;
}

let arr = [1, 2, 3, 4, 5]
let result = findEven(arr);
console.log(result);






//Approach-2
let arr = [1, 2, 3, 4, 5]
let result = arr.filter(function(num) {
    if (num % 2 == 0) {
        return num;
    }
});
console.log(result);



//Approach-3
let arr = [1, 2, 3, 4, 5]
let result = arr.filter(function(num) {
    return num % 2 == 0;
});
console.log(result);


//Approach-4
let arr = [1, 2, 3, 4, 5]
let result = arr.filter((num) => num % 2 == 0);
console.log(result);






Demo on reduce() function
*************************

//Normal Approach
function findSum(arr) {
    let sum = 0;
    for (let i = 0; i < arr.length; i++) {
        sum += arr[i];
    }
    return sum;
}

let arr = [1, 2, 3, 4, 5]
let result = findSum(arr);
console.log("Total: " + result);




//Approach-2
let arr = [1, 2, 3, 4, 5]
let result = arr.reduce(function(sum, num) {
    return sum + num;
}, 0);
console.log("Total: " + result);



//Approach-3
let arr = [1, 2, 3, 4, 5]
let result = arr.reduce((sum, num) => sum + num, 0);
console.log("Total: " + result);
