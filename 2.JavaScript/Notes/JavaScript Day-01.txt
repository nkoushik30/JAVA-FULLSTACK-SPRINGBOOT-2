Problem Statement:
If number is divisible by 3, display "Fizz", if number is divisible by 5, display "Buzz", if number is divisible by 3 and 5, display "Fizz Buzz".


<html lang="en">
<head>
    <title>JavaScript</title>
    
</head>
<body>
    <h1 id="result">Show Result Here</h1>

    <script>
        let num = 30;    //9 (Fizz), 25 (Buzz), 30 (Fizz Buzz)
        let result = "";
        
        if (num % 3 == 0)
            result = "Fizz ";

        if (num % 5 == 0)
            result += "Buzz";

        document.getElementById("result").innerHTML = result;
    </script>    
</body>
</html>

Prime Number
************
<html lang="en">
<head>
    <title>JavaScript</title>
    
</head>
<body>
    <h1 id="result">Show Result Here</h1>

    <script>
        let result = "";
        let num = 9;
        let count = 0;
        let i = 1;

        while (i <= num) {
            if (num % i == 0)
                count++;
            i++;
        }

        if (count == 2)
            result = "Prime Number";
        else
            result = "Not a Prime Number";

        document.getElementById("result").innerHTML = result;
    </script>    
</body>
</html>
